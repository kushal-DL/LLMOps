deque(maxlen=1000) as a metrics window Python's deque with maxlen automatically evicts the oldest entry when the capacity is reached — perfect for a rolling window of the last N requests. It's O(1) append and O(1) pop, unlike a list which would be O(n) for pop-left.

2. Model Validation & Compatibility

Before deploying any model, a ModelValidator runs a battery of automated checks. Anything that fails blocks the deploy — this is the first production gate in the pipeline.

class ModelValidator:
    def __init__(self):
        self.checks_passed = []
        self.checks_failed = []

    def validate_model_structure(self, model):
        """Check model is nn.Module with trainable parameters"""
        assert isinstance(model, nn.Module)
        assert sum(p.numel() for p in model.parameters()) > 0
        self.checks_passed.append("Model structure valid")

    def validate_forward_pass(self, model, input_shape, device):
        """Run a dummy forward pass, detect NaN/Inf"""
        dummy = torch.randn(input_shape).to(device)
        with torch.no_grad():
            out = model(dummy)
        assert not torch.isnan(out).any(), "NaN values in output"
        assert not torch.isinf(out).any(), "Inf values in output"
        self.checks_passed.append(f"Forward pass OK (output: {tuple(out.shape)})")

    def validate_determinism(self, model, input_shape, device, runs=3):
        """Same input produces same output across multiple runs"""
        x = torch.randn(input_shape).to(device)
        outs = [model(x).cpu() for _ in range(runs)]
        max_diff = max((outs[0] - outs[i]).abs().max() for i in range(1, runs))
        assert max_diff < 1e-5, f"Non-deterministic: max_diff={max_diff}"
        self.checks_passed.append("Deterministic outputs")

    def validate_memory(self, model, device):
        """Ensure model fits in 80% of available VRAM"""
        if torch.cuda.is_available():
            total = torch.cuda.get_device_properties(0).total_memory / 1e9
            used  = sum(p.numel() * 4 for p in model.parameters()) / 1e9
            assert used < total * 0.8, f"Too large: {used:.2f}GB / {total:.2f}GB"
        self.checks_passed.append("Memory OK")

    def get_report(self):
        status = "APPROVED" if not self.checks_failed else "NEEDS ATTENTION"
        return {'status': status,
                'passed': self.checks_passed,
                'failed': self.checks_failed}

Why check determinism? Batch norm and dropout layers behave differently in train() vs eval() mode. A model served without model.eval() will produce different outputs on identical inputs, making debugging impossible. Determinism checks surface this class of error before deploy.

3. Comprehensive Metrics Logger

The MetricsLogger maintains a bounded sliding window of the last 1000 requests. It records per-request latency, token counts, and cost, then computes percentile statistics on demand.

class MetricsLogger:
    def __init__(self, window_size=1000):
        self.metrics = defaultdict(lambda: deque(maxlen=window_size))
        self.start_time = datetime.now()

    def log_request(self, request_id, input_tokens, output_tokens, latency_ms, cost_usd):
        self.metrics['request_ids'].append(request_id)
        self.metrics['input_tokens'].append(input_tokens)
        self.metrics['output_tokens'].append(output_tokens)
        self.metrics['latency_ms'].append(latency_ms)
        self.metrics['cost_usd'].append(cost_usd)
        self.metrics['timestamp'].append(datetime.now())

    def get_percentile(self, values, p):
        sorted_vals = sorted(values)
        idx = int(len(sorted_vals) * p / 100)
        return sorted_vals[min(idx, len(sorted_vals) - 1)]

    def get_summary(self):
        latencies = list(self.metrics['latency_ms'])
        costs     = list(self.metrics['cost_usd'])
        tokens    = [i + o for i, o in zip(
                       self.metrics['input_tokens'], self.metrics['output_tokens'])]
        return {
            'requests_total':          len(latencies),
            'latency_p50':             self.get_percentile(latencies, 50),
            'latency_p95':             self.get_percentile(latencies, 95),
            'latency_p99':             self.get_percentile(latencies, 99),
            'cost_total_usd':          sum(costs),
            'cost_per_request':        sum(costs) / len(costs) if costs else 0,
            'throughput_tokens_per_sec': sum(tokens) / (sum(latencies) / 1000),
            'uptime':                  (datetime.now() - self.start_time).total_seconds() / 3600,
        }

4. Cost & Latency Dashboard

The Dashboard class renders the metrics summary into a human-readable view. In production this maps to a Grafana panel, Datadog dashboard, or custom web UI — the underlying summary dict is the same.

class Dashboard:
    def render_latency_histogram(self, latencies):
        bins = [0, 50, 100, 150, 200, 300, 500, 1000]
        counts = [0] * (len(bins) - 1)
        for lat in latencies:
            for i in range(len(bins) - 1):
                if bins[i] <= lat < bins[i + 1]:
                    counts[i] += 1
                    break
        return counts

    def render_dashboard(self, metrics_summary):
        print("┌─ PRODUCTION DASHBOARD ─────────────────────┐")
        print(f"│ Requests:   {metrics_summary['requests_total']:>30} │")
        print(f"│ Uptime (h): {metrics_summary['uptime']:>30.2f} │")
        print(f"│ P50:        {metrics_summary['latency_p50']:>30.1f} ms │")
        print(f"│ P95:        {metrics_summary['latency_p95']:>30.1f} ms │")
        print(f"│ P99:        {metrics_summary['latency_p99']:>30.1f} ms │")
        print(f"│ Tokens/s:   {metrics_summary['throughput_tokens_per_sec']:>30.1f} │")
        print(f"│ Cost/Req:   ${metrics_summary['cost_per_request']:>29.6f} │")
        print("│ STATUS: HEALTHY                            │")
        print("└────────────────────────────────────────────┘")

5. Safety Guardrails & Content Filtering

The SafetyGuardrails class implements three layers of protection: keyword-based input filtering, output pattern checking, and per-user rate limiting. These are entry and exit gates around every inference call.

class SafetyGuardrails:
    def __init__(self):
        self.blocked_keywords = [
            'malware', 'exploit', 'hack', 'crack',
            'illegal', 'bomb', 'violence', 'abuse'
        ]

    def check_input(self, text):
        """Block on keywords and length"""
        text_lower = str(text).lower()
        for kw in self.blocked_keywords:
            if kw in text_lower:
                return False, f"Blocked keyword: {kw}"
        if len(text) > 10000:
            return False, "Input exceeds max length"
        return True, "Input safe"

    def check_output(self, text):
        """Flag dangerous instruction patterns"""
        text_lower = str(text).lower()
        risky = ['how to make', 'instructions for', 'step by step']
        for pattern in risky:
            if pattern in text_lower:
                for kw in self.blocked_keywords:
                    if kw in text_lower:
                        return False, f"Pattern + keyword: '{pattern}' + '{kw}'"
        return True, "Output safe"

    def rate_limit_check(self, user_id, request_history, max_rpm=10):
        """Allow max 10 requests per user per minute"""
        now = datetime.now()
        one_min_ago = now - timedelta(minutes=1)
        recent = [t for t in request_history.get(user_id, []) if t > one_min_ago]
        if len(recent) >= max_rpm:
            return False, f"Rate limit: {len(recent)}/{max_rpm} rpm"
        return True, "Within limits"

Limitations of keyword-based filtering Keyword filters are fast and auditable, but brittle. Adversarial users can bypass them with synonyms ("expl0it", "h@ck"), base64 encoding, or role-play framing. Production systems layer multiple approaches: regex, classifier models (e.g. OpenAI Moderation API), and semantic similarity to known harmful prompts. The keyword filter here is a first line of defence, not a complete solution.

6. FastAPI Service Skeleton

The notebook prints a complete FastAPI service template — every component built in this session wired together into a production REST API. Three endpoints cover the minimal viable service: generate, health check, and metrics export.

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI(title="LLM Service", version="1.0.0")

class GenerationRequest(BaseModel):
    prompt: str
    max_tokens: int = 100
    temperature: float = 0.7

class GenerationResponse(BaseModel):
    generated_text: str
    tokens_generated: int
    latency_ms: float

@app.on_event("startup")
async def startup():
    global model, tokenizer, metrics_logger, guardrails
    # Load model and dependencies here
    pass

@app.post("/generate", response_model=GenerationResponse)
async def generate(request: GenerationRequest):
    # 1. Input safety gate
    is_safe, reason = guardrails.check_input(request.prompt)
    if not is_safe:
        raise HTTPException(status_code=400, detail=reason)

    start = time.time()

    # 2. Inference
    tokens = tokenizer.encode(request.prompt)
    with torch.no_grad():
        output = model.generate(tokens, max_tokens=request.max_tokens)
    generated_text = tokenizer.decode(output)

    # 3. Output safety gate
    is_safe, reason = guardrails.check_output(generated_text)
    if not is_safe:
        generated_text = "[Content filtered]"

    latency_ms = (time.time() - start) * 1000

    # 4. Log metrics
    metrics_logger.log_request(
        request_id=str(uuid.uuid4()),
        input_tokens=len(tokens),
        output_tokens=len(tokenizer.encode(generated_text)),
        latency_ms=latency_ms,
        cost_usd=0.00002,
    )
    return GenerationResponse(generated_text=generated_text,
                              tokens_generated=len(tokenizer.encode(generated_text)),
                              latency_ms=latency_ms)

@app.get("/health")
async def health():
    return {"status": "healthy", "model": "loaded"}

@app.get("/metrics")
async def get_metrics():
    return metrics_logger.get_summary()
